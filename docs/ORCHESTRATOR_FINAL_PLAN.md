## 🎮 Makefile (Clean & Simple)

```makefile
# Makefile

.PHONY: help setup prepare start stop restart logs status build config clean

# Load environment
-include .env
export

# Generate compose file list
COMPOSE_FILES := $(shell ./scripts/lib/list-compose-files.sh 2>/dev/null || echo "")
COMPOSE := docker-compose $(COMPOSE_FILES)

# Colors
RED := \033[0;31m
GREEN := \033[0;32m
YELLOW := \033[1;33m
NC := \033[0m # No Color

help:
	@echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
	@echo "  LG-Development Orchestrator"
	@echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
	@echo ""
	@echo "$(GREEN)Setup Commands:$(NC)"
	@echo "  make setup          - Interactive environment setup (.env)"
	@echo "  make prepare        - Clone all repos from GitHub"
	@echo ""
	@echo "$(GREEN)Development Commands:$(NC)"
	@echo "  make start          - Start all services (with dependencies)"
	@echo "  make stop           - Stop all services"
	@echo "  make restart        - Restart all services"
	@echo "  make logs [SERVICE] - View logs (all or specific)"
	@echo "  make status         - Show service status"
	@echo ""
	@echo "$(GREEN)Build Commands:$(NC)"
	@echo "  make build [SERVICE] - Build services"
	@echo "  make config          - Show final docker-compose config"
	@echo ""
	@echo "$(GREEN)Cleanup Commands:$(NC)"
	@echo "  make clean          - Stop services and remove containers"
	@echo "  make clean-repos    - Remove all cloned repos (⚠️  DANGER)"
	@echo ""

# Setup: Interactive .env generation
setup:
	@./scripts/setup.sh

# Prepare: Clone all repos
prepare:
	@if [ ! -f .env ]; then \
		echo "$(RED)❌ .env not found! Run 'make setup' first$(NC)"; \
		exit 1; \
	fi
	@./scripts/prepare.sh

# Start with dependency resolution
start:
	@if [ -z "$(COMPOSE_FILES)" ]; then \
		echo "$(RED)❌ No repos found! Run 'make prepare' first$(NC)"; \
		exit 1; \
	fi
	@./scripts/start.sh

# Stop all services
stop:
	@if [ -n "$(COMPOSE_FILES)" ]; then \
		echo "$(YELLOW)🛑 Stopping services...$(NC)"; \
		$(COMPOSE) down; \
	else \
		echo "$(YELLOW)⊘ No services running$(NC)"; \
	fi

# Restart
restart: stop start

# Logs
logs:
ifdef SERVICE
	@$(COMPOSE) logs -f $(SERVICE)
else
	@$(COMPOSE) logs -f
endif

# Status
status:
	@./scripts/status.sh

# Build
build:
ifdef SERVICE
	@$(COMPOSE) build $(SERVICE)
else
	@$(COMPOSE) build
endif

# Show final config (debugging)
config:
	@$(COMPOSE) config

# Clean
clean:
	@echo "$(YELLOW)🧹 Cleaning up...$(NC)"
	@$(COMPOSE) down -v 2>/dev/null || true
	@echo "$(GREEN)✅ Cleanup complete$(NC)"

clean-repos:
	@echo "$(RED)⚠️  This will DELETE all cloned repos in repos/$(NC)"
	@read -p "Are you sure? (yes/no) " confirm && [ "$$confirm" = "yes" ] || exit 1
	@rm -rf repos/
	@echo "$(GREEN)✅ Repos removed$(NC)"
```

## 🔧 Script: start.sh (With Dependencies)

```bash
#!/bin/bash
# scripts/start.sh
#
# Start services with dependency resolution

set -e

source scripts/lib/common.sh
source .env

print_header "Starting Services"

# Build dependency graph
print_step "Analyzing dependencies..."
./scripts/lib/dependency-graph.sh > /tmp/lg-dep-graph.txt

# Show graph
cat /tmp/lg-dep-graph.txt

# Determine start order
print_step "Determining start order..."

# Layer 0: Infrastructure (no dependencies)
LAYER_0=(
    "repos/lg-traefik/docker-compose.yml"
    "repos/lg-postgres/docker-compose.yml"
    "repos/lg-redis/docker-compose.yml"
    "repos/lg-dynamodb/docker-compose.yml"
    "repos/lg-verdaccio/docker-compose.yml"
)

# Layer 1: Services (depend on infrastructure)
LAYER_1=()
for service_dir in repos/lg-*-service/; do
    if [ -f "${service_dir}docker-compose.yml" ]; then
        LAYER_1+=("${service_dir}docker-compose.yml")
    fi
done

# Layer 2: Frontend (depends on services)
LAYER_2=()
if [ -f "repos/lg-admin/docker-compose.yml" ]; then
    LAYER_2+=("repos/lg-admin/docker-compose.yml")
fi

# Build compose command
COMPOSE_FILES=""
for file in "${LAYER_0[@]}" "${LAYER_1[@]}" "${LAYER_2[@]}"; do
    if [ -f "$file" ]; then
        COMPOSE_FILES="$COMPOSE_FILES -f $file"
    fi
done

COMPOSE="docker-compose $COMPOSE_FILES"

# Start
print_step "Starting services..."
echo ""
echo "Command: $COMPOSE up -d"
echo ""

$COMPOSE up -d

# Wait for health
print_step "Waiting for services to be healthy..."
./scripts/lib/wait-for-health.sh

# Show status
print_step "Service Status"
$COMPOSE ps

success "All services started!"
echo ""
echo "Access:"
if [ "$USE_PROXY_DOMAINS" = "true" ]; then
    echo "  Admin UI:   http://admin.lg.local"
    echo "  API:        http://api.lg.local/user/health"
    echo "  Traefik:    http://traefik.lg.local"
else
    echo "  Admin UI:   http://localhost"
    echo "  API:        http://localhost/api/user/health"
    echo "  Traefik:    http://localhost:8080"
fi
echo ""
echo "View logs: make logs"
```

### docker-compose.yml
```yaml
version: '3.8'

services:
  postgres:
    container_name: lg-infra-postgres
    image: postgres:16-alpine
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-licenseguard}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-changeme}
      POSTGRES_DB: ${POSTGRES_DB:-licenseguard}
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql:ro
    networks:
      - lg-internal
    ports:
      - "5432:5432"  # Optional: For local access
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-licenseguard}"]
      interval: 10s
      timeout: 5s
      retries: 5

networks:
  lg-internal:
    external: true

volumes:
```
